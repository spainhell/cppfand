{ nová věta na místě, kde na disku zůstala dřívější věta: po nrecs:=0 (texty jsou pryč)
  a po deleterec (v souboru bez indexu zůstane za poslední větou její kopie s pozicemi
  jejích textů); zápis nové věty nesmí texty dřívější věty číst ani rušit (WriteRec v PC-FANDu);
  soubor bez indexu, indexovaný soubor a .DBF; zvětšení nrecs }
VAR a:record of TNOV; x:record of TNOVX; d:record of TNOVD; i:real; chyba:string;
begin
chyba:='';
{ 1. nrecs:=0 a nová věta s textem }
TNOV.nrecs:=0; TNOVX.nrecs:=0; TNOVD.nrecs:=0;
for i:=1 to 3 do begin
  readrec(a,0); a.Id:=i; a.Pozn:=repeatstr('a'+str(i,1,0)+' ',i*200); writerec(a,0);
  readrec(x,0); x.Kod:='K'+str(i,1,0); x.Pozn:=repeatstr('x'+str(i,1,0)+' ',i*200); writerec(x,0);
  readrec(d,0); d.Id:=i; d.Pozn:=repeatstr('d'+str(i,1,0)+' ',i*200); writerec(d,0);
end;
TNOV.nrecs:=0; TNOVX.nrecs:=0; TNOVD.nrecs:=0;
readrec(a,0); a.Id:=9; a.Pozn:='nová a'; writerec(a,0);
readrec(x,0); x.Kod:='K9'; x.Pozn:='nová x'; writerec(x,0);
readrec(d,0); d.Id:=9; d.Pozn:='nová d'; writerec(d,0);
close(TNOV); close(TNOVX); close(TNOVD);
readrec(a,1);
if (chyba='') & ((TNOV.nrecs<>1) | (a.Id<>9) | (a.Pozn<>'nová a')) then chyba:='TNOV po nrecs:=0 nrecs='+str(TNOV.nrecs,1,0)+' '+a.Pozn;
readrec(x,keyof(TNOVX,'K9'));
if (chyba='') & ((TNOVX.nrecs<>1) | (x.Pozn<>'nová x')) then chyba:='TNOVX po nrecs:=0 nrecs='+str(TNOVX.nrecs,1,0)+' '+x.Pozn;
readrec(d,1);
if (chyba='') & ((TNOVD.nrecs<>1) | (d.Id<>9) | (d.Pozn<>'nová d')) then chyba:='TNOVD po nrecs:=0 nrecs='+str(TNOVD.nrecs,1,0)+' '+d.Pozn;

{ 2. deleterec a nová věta: texty ostatních vět zůstanou }
TNOV.nrecs:=0; TNOVX.nrecs:=0; TNOVD.nrecs:=0;
for i:=1 to 5 do begin
  readrec(a,0); a.Id:=i; a.Pozn:=repeatstr('a'+str(i,1,0)+' ',i*150); writerec(a,0);
  readrec(x,0); x.Kod:='K'+str(i,1,0); x.Pozn:=repeatstr('x'+str(i,1,0)+' ',i*150); writerec(x,0);
  readrec(d,0); d.Id:=i; d.Pozn:=repeatstr('d'+str(i,1,0)+' ',i*150); writerec(d,0);
end;
deleterec(TNOV,2); deleterec(TNOVX,keyof(TNOVX,'K2')); deleterec(TNOVD,2);
readrec(a,0); a.Id:=6; a.Pozn:=repeatstr('a6 ',900); writerec(a,0);
readrec(x,0); x.Kod:='K6'; x.Pozn:=repeatstr('x6 ',900); writerec(x,0);
readrec(d,0); d.Id:=6; d.Pozn:=repeatstr('d6 ',900); writerec(d,0);
close(TNOV); close(TNOVX); close(TNOVD);
if (chyba='') & (TNOV.nrecs<>5) then chyba:='TNOV po deleterec nrecs='+str(TNOV.nrecs,1,0);
for i:=1 to 5 do begin
  readrec(a,i);
  if (chyba='') & (a.Pozn<>repeatstr('a'+str(a.Id,1,0)+' ',a.Id*150)) then chyba:='TNOV po deleterec text věty Id '+str(a.Id,1,0);
end;
for i:=1 to 6 do if i<>2 then begin
  readrec(x,keyof(TNOVX,'K'+str(i,1,0)));
  if (chyba='') & (x.Pozn<>repeatstr('x'+str(i,1,0)+' ',i*150)) then chyba:='TNOVX po deleterec text K'+str(i,1,0);
end;
if (chyba='') & (TNOVD.nrecs<>5) then chyba:='TNOVD po deleterec nrecs='+str(TNOVD.nrecs,1,0);
for i:=1 to 5 do begin
  readrec(d,i);
  if (chyba='') & (d.Pozn<>repeatstr('d'+str(d.Id,1,0)+' ',d.Id*150)) then chyba:='TNOVD po deleterec text věty Id '+str(d.Id,1,0);
end;

{ 3. nrecs:=0 a zvětšení nrecs: přidají se prázdné (zrušené) věty }
TNOV.nrecs:=0; TNOVD.nrecs:=0;
TNOV.nrecs:=3; TNOVD.nrecs:=3;
readrec(a,3); a.Id:=3; a.Pozn:='po zvětšení'; writerec(a,3);
readrec(d,3); d.Id:=3; d.Pozn:='po zvětšení'; writerec(d,3);
close(TNOV); close(TNOVD);
readrec(a,3);
if (chyba='') & ((TNOV.nrecs<>3) | (a.Pozn<>'po zvětšení')) then chyba:='TNOV zvětšení nrecs='+str(TNOV.nrecs,1,0)+' '+a.Pozn;
readrec(a,1);
if (chyba='') & (a.Pozn<>'') then chyba:='TNOV zvětšení věta 1 má text';
readrec(d,3);
if (chyba='') & ((TNOVD.nrecs<>3) | (d.Pozn<>'po zvětšení')) then chyba:='TNOVD zvětšení nrecs='+str(TNOVD.nrecs,1,0)+' '+d.Pozn;
proc(Vysl,('NovaVeta',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
