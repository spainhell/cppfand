{ druhý uživatel testu Viditelnost }
VAR r:record of TVID; i,k:real; ok:boolean; t:string;
begin
proc(Cekej,('A','vid-zapsano',ok));
if TVID.nrecs<>10 then proc(Signal,('B','Viditelnost!','nrecs='+str(TVID.nrecs,1,0)));
for i:=1 to TVID.nrecs do begin
  readrec(r,i);
  if (r.Kod<>i) | (r.Pozn<>repeatstr(char(64+i),100*i)) then proc(Signal,('B','Viditelnost!','věta '+str(i,1,0)+' od A'));
end;
readrec(r,5);
r.Nazev:='B5';
r.Pozn:=repeatstr('b',3000);
writerec(r,5);
readrec(r,0);
r.Kod:=11;
r.Nazev:='B11';
r.Pozn:=repeatstr('c',2000);
writerec(r,0);
deleterec(TVID,2);
proc(Signal,('B','vid-zmeneno',''));
proc(Cekej,('A','vid-pridano',ok));
if TVID.nrecs<>11 then proc(Signal,('B','Viditelnost!','nrecs po přidání A='+str(TVID.nrecs,1,0)));
for i:=1 to TVID.nrecs do begin
  readrec(r,i);
  k:=cond((i=1):1,(i=10):11,(i=11):12,else:i+1);
  t:=cond((k=5):repeatstr('b',3000),(k=11):repeatstr('c',2000),(k=12):repeatstr('d',1500),else:repeatstr(char(64+k),100*k));
  if (r.Kod<>k) | (r.Pozn<>t) then proc(Signal,('B','Viditelnost!','po přidání A věta '+str(i,1,0)+' Kod '+str(r.Kod,1,0)));
end;
proc(Signal,('B','Viditelnost-konec',''));
end;
