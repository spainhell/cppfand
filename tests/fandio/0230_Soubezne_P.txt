{ oba uživatelé současně přidávají věty s texty do indexovaného souboru, souboru
  bez indexu a .DBF; žádná věta ani text se nesmí ztratit ani přepsat }
VAR r:record of TSOUB; n:record of TSOUBN; d:record of TSOUBD;
    i,k,na,nb,p,c:real; ok:boolean; chyba,t:string;
begin
chyba:='';
TSOUB.nrecs:=0;
TSOUBN.nrecs:=0;
TSOUBD.nrecs:=0;
proc(Signal,('A','sou-start',''));
proc(Cekej,('B','sou-start',ok));
for i:=1 to 150 do begin
  k:=0+i;
  t:=repeatstr('a',(k-int(k/7)*7)*300+1);
  readrec(r,0);
  r.Kdo:='A';
  r.Kod:=k;
  r.Pozn:=t;
  writerec(r,0);
  readrec(n,0);
  n.Kdo:='A';
  n.Kod:=k;
  n.Pozn:=t;
  writerec(n,0);
  readrec(d,0);
  d.Kdo:='A';
  d.Kod:=k;
  d.Pozn:=t;
  writerec(d,0);
end;
proc(Signal,('A','sou-hotovo',''));
proc(Cekej,('B','sou-hotovo',ok));
if ^ok then chyba:='druhý uživatel nedoběhl';
na:=0; nb:=0;
for i:=1 to TSOUB.nrecs do begin
  readrec(r,i);
  k:=r.Kod;
  t:=repeatstr(cond((trailchar(' ',r.Kdo)='A'):'a',else:'b'),(k-int(k/7)*7)*300+1);
  if (trailchar(' ',r.Kdo)='A') & (k>=1) & (k<=150) then na:=na+k
  else if (trailchar(' ',r.Kdo)='B') & (k>=1001) & (k<=1150) then nb:=nb+k
  else if chyba='' then chyba:='TSOUB věta '+str(i,1,0)+': Kdo '+r.Kdo+' Kod '+str(k,1,0);
  if (chyba='') & (r.Pozn<>t) then chyba:='TSOUB text věty '+str(i,1,0)+' (Kod '+str(k,1,0)+')';
end;
if (chyba='') & ((TSOUB.nrecs<>300) | (na<>11325) | (nb<>161325)) then chyba:='TSOUB nrecs='+str(TSOUB.nrecs,1,0)+' součty '+str(na,1,0)+'/'+str(nb,1,0);
na:=0; nb:=0;
for i:=1 to TSOUBN.nrecs do begin
  readrec(n,i);
  k:=n.Kod;
  t:=repeatstr(cond((trailchar(' ',n.Kdo)='A'):'a',else:'b'),(k-int(k/7)*7)*300+1);
  if (trailchar(' ',n.Kdo)='A') & (k>=1) & (k<=150) then na:=na+k
  else if (trailchar(' ',n.Kdo)='B') & (k>=1001) & (k<=1150) then nb:=nb+k
  else if chyba='' then chyba:='TSOUBN věta '+str(i,1,0)+': Kdo '+n.Kdo+' Kod '+str(k,1,0);
  if (chyba='') & (n.Pozn<>t) then chyba:='TSOUBN text věty '+str(i,1,0)+' (Kod '+str(k,1,0)+')';
end;
if (chyba='') & ((TSOUBN.nrecs<>300) | (na<>11325) | (nb<>161325)) then chyba:='TSOUBN nrecs='+str(TSOUBN.nrecs,1,0)+' součty '+str(na,1,0)+'/'+str(nb,1,0);
na:=0; nb:=0;
for i:=1 to TSOUBD.nrecs do begin
  readrec(d,i);
  k:=d.Kod;
  t:=repeatstr(cond((trailchar(' ',d.Kdo)='A'):'a',else:'b'),(k-int(k/7)*7)*300+1);
  if (trailchar(' ',d.Kdo)='A') & (k>=1) & (k<=150) then na:=na+k
  else if (trailchar(' ',d.Kdo)='B') & (k>=1001) & (k<=1150) then nb:=nb+k
  else if chyba='' then chyba:='TSOUBD věta '+str(i,1,0)+': Kdo '+d.Kdo+' Kod '+str(k,1,0);
  if (chyba='') & (d.Pozn<>t) then chyba:='TSOUBD text věty '+str(i,1,0)+' (Kod '+str(k,1,0)+')';
end;
if (chyba='') & ((TSOUBD.nrecs<>300) | (na<>11325) | (nb<>161325)) then chyba:='TSOUBD nrecs='+str(TSOUBD.nrecs,1,0)+' součty '+str(na,1,0)+'/'+str(nb,1,0);
{ index: vzestupně podle Kod, všechny věty }
c:=0; p:=0;
forall r/@ do begin
  if (chyba='') & (r.Kod<=p) then chyba:='index TSOUB: pořadí u Kod '+str(r.Kod,1,0);
  p:=r.Kod;
  c:=c+1;
end;
if (chyba='') & (c<>300) then chyba:='index TSOUB: forall prošel '+str(c,1,0)+' vět';
proc(ChybaB,('Soubezne',chyba));
proc(Vysl,('Soubezne',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
