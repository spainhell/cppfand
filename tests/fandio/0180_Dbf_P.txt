{ soubor .DBF: zápis a čtení všech typů údajů a memo textů, přepis, zrušení věty
  (následující se posunou), appendrec, čtení po zavření souboru }
VAR r:record of TDBF; r2:record of TDBF2; r3:record of TDBF3; rc:record of TDBFC; i,n,s,d0:real; chyba,t:string;
begin
chyba:='';
TDBF.nrecs:=0;
d0:=valdate('01.03.2025','DD.MM.YYYY');
for i:=1 to 40 do begin
  n:=cond(((i-int(i/5)*5)=0):0,((i-int(i/5)*5)=1):10,((i-int(i/5)*5)=2):511,((i-int(i/5)*5)=3):1000,else:5000);
  readrec(r,0);
  r.Kod:=i;
  r.Nazev:='Položka '+str(i,2,0);
  r.Cena:=i*12.34;
  r.Datum:=d0+i;
  r.Platny:=(i-int(i/3)*3)=0;
  r.Pozn:=copy(repeatstr(char(64+i)+'bcdefghij',int(n/10)+1),1,n);
  writerec(r,0);
end;
if TDBF.nrecs<>40 then chyba:='nrecs='+str(TDBF.nrecs,1,0);
close(TDBF);
for i:=1 to 40 do begin
  n:=cond(((i-int(i/5)*5)=0):0,((i-int(i/5)*5)=1):10,((i-int(i/5)*5)=2):511,((i-int(i/5)*5)=3):1000,else:5000);
  readrec(r,i);
  t:=copy(repeatstr(char(64+i)+'bcdefghij',int(n/10)+1),1,n);
  if chyba='' then begin
    if r.Kod<>i then chyba:='Kod, věta '+str(i,1,0)
    else if trailchar(' ',r.Nazev)<>'Položka '+str(i,2,0) then chyba:='Nazev, věta '+str(i,1,0)+': '+r.Nazev
    else if abs(r.Cena-i*12.34)>0.001 then chyba:='Cena, věta '+str(i,1,0)+': '+str(r.Cena,10,2)
    else if r.Datum<>d0+i then chyba:='Datum, věta '+str(i,1,0)
    else if (r.Platny & ((i-int(i/3)*3)<>0)) | (^r.Platny & ((i-int(i/3)*3)=0)) then chyba:='Platny, věta '+str(i,1,0)
    else if length(r.Pozn)<>n then chyba:='délka textu, věta '+str(i,1,0)+': '+str(length(r.Pozn),1,0)
    else if r.Pozn<>t then chyba:='obsah textu, věta '+str(i,1,0);
  end;
end;
{ přepis: každá čtvrtá věta dostane jiný název, cenu a delší text }
for i:=1 to 10 do begin
  readrec(r,i*4);
  r.Nazev:='Změna '+str(i*4,2,0);
  r.Cena:=-r.Cena;
  r.Pozn:=repeatstr('Z',700+i);
  writerec(r,i*4);
end;
close(TDBF);
for i:=1 to 10 do begin
  readrec(r,i*4);
  if (chyba='') & ((trailchar(' ',r.Nazev)<>'Změna '+str(i*4,2,0)) | (abs(r.Cena+i*4*12.34)>0.001)) then chyba:='po přepisu věta '+str(i*4,1,0)
  else if (chyba='') & (r.Pozn<>repeatstr('Z',700+i)) then chyba:='po přepisu text věty '+str(i*4,1,0);
end;
{ zrušení 2. věty: následující se posunou }
deleterec(TDBF,2);
close(TDBF);
if (chyba='') & (TDBF.nrecs<>39) then chyba:='po rušení nrecs='+str(TDBF.nrecs,1,0);
readrec(r,2);
if (chyba='') & (r.Kod<>3) then chyba:='po rušení věta 2 má Kod '+str(r.Kod,1,0);
readrec(r,3);
if (chyba='') & (r.Pozn<>repeatstr('Z',701)) then chyba:='po rušení text věty 3 (Kod 4)';
{ appendrec }
appendrec(TDBF);
readrec(r,40);
r.Kod:=99; r.Nazev:='Nová'; r.Pozn:='memo nové věty';
writerec(r,40);
close(TDBF);
readrec(r,40);
if (chyba='') & ((TDBF.nrecs<>40) | (r.Kod<>99) | (r.Pozn<>'memo nové věty')) then chyba:='appendrec';
{ forall přes .DBF: zrušená věta z appendrec se přeskočí }
n:=0; s:=0;
forall r do begin n:=n+1; s:=s+r.Kod; end;
if (chyba='') & ((n<>39) | (s<>818)) then chyba:='forall prošel '+str(n,1,0)+' vět, součet Kod '+str(s,1,0);
{ merge do .DBF: platné věty mají Kod 3,6,..,39 }
merge(['#I1_ TDBF (Platny) #O1_ TDBF2']);
n:=0; s:=0;
forall r2 do begin
  n:=n+1; s:=s+r2.Kod;
  if (chyba='') & (r2.Kod=3) & (r2.Pozn<>copy(repeatstr('Cbcdefghij',101),1,1000)) then chyba:='merge: text věty Kod 3';
  if (chyba='') & (r2.Kod=12) & (r2.Pozn<>repeatstr('Z',703)) then chyba:='merge: text věty Kod 12';
end;
if (chyba='') & ((TDBF2.nrecs<>13) | (n<>13) | (s<>273)) then chyba:='merge nrecs='+str(TDBF2.nrecs,1,0)+' součet Kod '+str(s,1,0);
merge(['#I1_ TDBF (Platny) #O1_ TDBF2 +']);
if (chyba='') & (TDBF2.nrecs<>26) then chyba:='merge + nrecs='+str(TDBF2.nrecs,1,0);
merge(['#I1_ TDBF ((Kod>30) & (Kod<50)) #O1_ TDBF2']);
readrec(r2,1);
if (chyba='') & ((TDBF2.nrecs<>10) | (r2.Kod<>31)) then chyba:='merge přepis nrecs='+str(TDBF2.nrecs,1,0);
{ merge se vstupem totožným s výstupem: TDBF2 má Kod 31..40 }
merge(['#I1_ TDBF2 #O1_ TDBF2 Kod:=Kod+1000']);
n:=0;
forall r2 do begin
  n:=n+1;
  if (chyba='') & (r2.Kod=1032) & (r2.Pozn<>repeatstr('Z',708)) then chyba:='merge do sebe: text Kod 1032';
  if (chyba='') & (r2.Kod=1034) & (length(r2.Pozn)<>5000) then chyba:='merge do sebe: text Kod 1034';
  if (chyba='') & ((r2.Kod<1031) | (r2.Kod>1040)) then chyba:='merge do sebe: Kod '+str(r2.Kod,1,0);
end;
if (chyba='') & ((n<>10) | (TDBF2.nrecs<>10)) then chyba:='merge do sebe nrecs='+str(TDBF2.nrecs,1,0);
{ sort sestupně: texty se musí přesunout s větou }
sort(TDBF2,(>Kod));
s:=10000;
for i:=1 to 10 do begin
  readrec(r2,i);
  if (chyba='') & (r2.Kod>=s) then chyba:='sort, věta '+str(i,1,0);
  if (chyba='') & (r2.Kod=1040) & (r2.Pozn<>repeatstr('Z',710)) then chyba:='sort: text Kod 1040';
  if (chyba='') & (r2.Kod=1034) & (length(r2.Pozn)<>5000) then chyba:='sort: text Kod 1034';
  s:=r2.Kod;
end;
readrec(r2,1);
if (chyba='') & ((TDBF2.nrecs<>10) | (r2.Kod<>1040)) then chyba:='sort nrecs='+str(TDBF2.nrecs,1,0);
{ merge s tříděním vstupu (pracovní index nad .DBF) }
merge(['#I1_ TDBF2 ! Kod #O1_ TDBF3']);
s:=0;
for i:=1 to TDBF3.nrecs do begin
  readrec(r3,i);
  if (chyba='') & (r3.Kod<=s) then chyba:='merge s tříděním, věta '+str(i,1,0);
  if (chyba='') & (r3.Kod=1032) & (r3.Pozn<>repeatstr('Z',708)) then chyba:='merge s tříděním: text Kod 1032';
  s:=r3.Kod;
end;
if (chyba='') & (TDBF3.nrecs<>10) then chyba:='merge s tříděním nrecs='+str(TDBF3.nrecs,1,0);
{ vazba do .DBF číselníku: hledání podle vlastního klíče v souboru bez indexu }
TDBFC.nrecs:=0;
for i:=1 to 5 do begin
  readrec(rc,0);
  rc.Kod:=1030+2*i;
  rc.Nazev:='C'+str(1030+2*i,4,0);
  writerec(rc,0);
end;
n:=0;
for i:=1 to TDBF3.nrecs do begin
  readrec(r3,i);
  if r3.TDBFC.exist then begin
    n:=n+1;
    if (chyba='') & (trailchar(' ',r3.TDBFC.Nazev)<>'C'+str(r3.Kod,4,0)) then chyba:='vazba u Kod '+str(r3.Kod,1,0);
  end;
end;
if (chyba='') & (n<>5) then chyba:='vazba našla '+str(n,1,0)+' vět';
proc(Vysl,('Dbf',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
