{ sort vzestupně a sestupně; ostatní údaje se musí přesunout s větou }
VAR r:record of TSORT; i,v:real; chyba:string;
begin
chyba:='';
TSORT.nrecs:=0;
{ čísla 1..52 v zamíchaném pořadí (permutace i*7 mod 53) }
for i:=1 to 50 do begin
  v:=i*7-int(i*7/53)*53;
  readrec(r,0);
  r.Cislo:=v;
  r.Jmeno:='J'+str(v,2,0);
  writerec(r,0);
end;
sort(TSORT,(Cislo));
if TSORT.nrecs<>50 then chyba:='nrecs='+str(TSORT.nrecs,1,0);
v:=0;
for i:=1 to 50 do begin
  readrec(r,i);
  if (chyba='') & (r.Cislo<=v) then chyba:='vzestupně, věta '+str(i,1,0);
  if (chyba='') & (trailchar(' ',r.Jmeno)<>'J'+str(r.Cislo,2,0)) then chyba:='Jmeno se nepřesunulo, věta '+str(i,1,0);
  v:=r.Cislo;
end;
sort(TSORT,(>Cislo));
v:=1000;
for i:=1 to 50 do begin
  readrec(r,i);
  if (chyba='') & (r.Cislo>=v) then chyba:='sestupně, věta '+str(i,1,0);
  if (chyba='') & (trailchar(' ',r.Jmeno)<>'J'+str(r.Cislo,2,0)) then chyba:='sestupně Jmeno, věta '+str(i,1,0);
  v:=r.Cislo;
end;
proc(Vysl,('Trideni',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
