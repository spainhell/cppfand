{ volné texty: délky kolem hranic stránky 512 B až po 65 000 B, přepis, rušení věty,
  čtení po zavření souboru }
VAR r:record of TTXT; i,n,k:real; chyba,t:string;
begin
chyba:='';
TTXT.nrecs:=0;
{ věta i: text z opakovaného vzoru s písmenem podle i, délka podle tabulky }
for i:=1 to 12 do begin
  n:=cond((i=1):0,(i=2):1,(i=3):100,(i=4):509,(i=5):510,(i=6):511,(i=7):512,(i=8):513,(i=9):1000,(i=10):4096,(i=11):20000,else:65000);
  readrec(r,0);
  r.Id:=i;
  r.Pozn:=copy(repeatstr(char(64+i)+'123456789',int(n/10)+1),1,n);
  writerec(r,0);
end;
close(TTXT);
for i:=1 to 12 do begin
  n:=cond((i=1):0,(i=2):1,(i=3):100,(i=4):509,(i=5):510,(i=6):511,(i=7):512,(i=8):513,(i=9):1000,(i=10):4096,(i=11):20000,else:65000);
  readrec(r,i);
  t:=copy(repeatstr(char(64+i)+'123456789',int(n/10)+1),1,n);
  if (chyba='') & (length(r.Pozn)<>n) then chyba:='délka textu věty '+str(i,1,0)+': '+str(length(r.Pozn),1,0)
  else if (chyba='') & (r.Pozn<>t) then chyba:='obsah textu věty '+str(i,1,0);
end;
{ přepis: liché věty delším textem, sudé kratším }
for i:=1 to 12 do begin
  readrec(r,i);
  k:=cond(((i-int(i/2)*2)=1):length(r.Pozn)+700,else:int(length(r.Pozn)/3));
  if k>65000 then k:=65000;
  r.Pozn:=repeatstr('x',k);
  writerec(r,i);
end;
{ zrušení 3. věty (neindexovaný soubor: následující se posunou) }
deleterec(TTXT,3);
close(TTXT);
if (chyba='') & (TTXT.nrecs<>11) then chyba:='po rušení nrecs='+str(TTXT.nrecs,1,0);
for i:=1 to 11 do begin
  readrec(r,i);
  n:=cond((r.Id=1):0,(r.Id=2):1,(r.Id=3):100,(r.Id=4):509,(r.Id=5):510,(r.Id=6):511,(r.Id=7):512,(r.Id=8):513,(r.Id=9):1000,(r.Id=10):4096,(r.Id=11):20000,else:65000);
  k:=cond(((r.Id-int(r.Id/2)*2)=1):n+700,else:int(n/3));
  if k>65000 then k:=65000;
  if (chyba='') & (r.Id<>cond((i<3):i,else:i+1)) then chyba:='po rušení věta '+str(i,1,0)+' má Id '+str(r.Id,1,0)
  else if (chyba='') & (r.Pozn<>repeatstr('x',k)) then chyba:='po přepisu text věty Id '+str(r.Id,1,0)+' délka '+str(length(r.Pozn),1,0);
end;
{ appendrec: nová věta na konci, text zapsaný dodatečně }
appendrec(TTXT);
if (chyba='') & (TTXT.nrecs<>12) then chyba:='po appendrec nrecs='+str(TTXT.nrecs,1,0);
readrec(r,12);
r.Id:=99;
r.Pozn:=repeatstr('z',777);
writerec(r,12);
close(TTXT);
readrec(r,12);
if (chyba='') & ((r.Id<>99) | (r.Pozn<>repeatstr('z',777))) then chyba:='appendrec věta 12';
readrec(r,1);
if (chyba='') & (r.Pozn<>repeatstr('x',700)) then chyba:='po appendrec text věty 1';
{ vyprázdnění (nrecs:=0) a nové věty: za koncem souboru zůstaly staré věty s pozicemi
  textů, nový zápis je nesmí brát jako původní obsah věty }
TTXT.nrecs:=0;
for i:=1 to 3 do begin
  readrec(r,0);
  r.Id:=i;
  r.Pozn:='nový '+str(i,1,0);
  writerec(r,0);
end;
close(TTXT);
if (chyba='') & (TTXT.nrecs<>3) then chyba:='po vyprázdnění nrecs='+str(TTXT.nrecs,1,0);
for i:=1 to TTXT.nrecs do begin
  readrec(r,i);
  if (chyba='') & ((r.Id<>i) | (r.Pozn<>'nový '+str(i,1,0))) then chyba:='po vyprázdnění věta '+str(i,1,0);
end;
proc(Vysl,('Texty',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
