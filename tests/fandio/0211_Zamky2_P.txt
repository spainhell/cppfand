{ druhý uživatel testu Zamky }
VAR ok,b:boolean;
begin
proc(Cekej,('A','Zamky-excl',ok));
b:=false;
with shared TZAM(RD) do b:=true else b:=false;
if b then proc(Signal,('B','Zamky!','RD při cizím EXCL'));
b:=false;
with shared TZAM(WR) do b:=true else b:=false;
if b then proc(Signal,('B','Zamky!','WR při cizím EXCL'));
proc(Signal,('B','Zamky-excl-ok',''));
proc(Cekej,('A','Zamky-excl-konec',ok));
b:=false;
with shared TZAM(RD) do b:=true else b:=false;
if ^b then proc(Signal,('B','Zamky!','RD po uvolnění EXCL'));
proc(Signal,('B','Zamky-excl-konec-ok',''));
proc(Cekej,('A','Zamky-rd',ok));
b:=false;
with shared TZAM(RD) do b:=true else b:=false;
if ^b then proc(Signal,('B','Zamky!','RD při cizím RD'));
b:=false;
with shared TZAM(WR) do b:=true else b:=false;
if b then proc(Signal,('B','Zamky!','WR při cizím RD'));
b:=false;
with shared TZAM(EXCL) do b:=true else b:=false;
if b then proc(Signal,('B','Zamky!','EXCL při cizím RD'));
proc(Signal,('B','Zamky-rd-ok',''));
proc(Cekej,('A','Zamky-veta',ok));
b:=false;
with locked TZAM[3] do b:=true else b:=false;
if b then proc(Signal,('B','Zamky!','zámek věty 3 při cizím zámku'));
b:=false;
with locked TZAM[4] do b:=true else b:=false;
if ^b then proc(Signal,('B','Zamky!','zámek věty 4 při cizím zámku věty 3'));
proc(Signal,('B','Zamky-veta-ok',''));
proc(Cekej,('A','Zamky-veta-konec',ok));
b:=false;
with locked TZAM[3] do b:=true else b:=false;
if ^b then proc(Signal,('B','Zamky!','zámek věty 3 po uvolnění'));
proc(Signal,('B','Zamky-konec',''));
end;
