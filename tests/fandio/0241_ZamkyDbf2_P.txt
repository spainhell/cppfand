{ druhý uživatel testu ZamkyDbf }
VAR ok,b:boolean;
begin
proc(Cekej,('A','ZamkyDbf-excl',ok));
b:=false;
with shared TZAMD(RD) do b:=true else b:=false;
if b then proc(Signal,('B','ZamkyDbf!','RD při cizím EXCL'));
b:=false;
with shared TZAMD(WR) do b:=true else b:=false;
if b then proc(Signal,('B','ZamkyDbf!','WR při cizím EXCL'));
proc(Signal,('B','ZamkyDbf-excl-ok',''));
proc(Cekej,('A','ZamkyDbf-excl-konec',ok));
b:=false;
with shared TZAMD(RD) do b:=true else b:=false;
if ^b then proc(Signal,('B','ZamkyDbf!','RD po uvolnění EXCL'));
proc(Signal,('B','ZamkyDbf-excl-konec-ok',''));
proc(Cekej,('A','ZamkyDbf-rd',ok));
b:=false;
with shared TZAMD(RD) do b:=true else b:=false;
if ^b then proc(Signal,('B','ZamkyDbf!','RD při cizím RD'));
b:=false;
with shared TZAMD(WR) do b:=true else b:=false;
if b then proc(Signal,('B','ZamkyDbf!','WR při cizím RD'));
b:=false;
with shared TZAMD(EXCL) do b:=true else b:=false;
if b then proc(Signal,('B','ZamkyDbf!','EXCL při cizím RD'));
proc(Signal,('B','ZamkyDbf-rd-ok',''));
proc(Cekej,('A','ZamkyDbf-veta',ok));
b:=false;
with locked TZAMD[3] do b:=true else b:=false;
if b then proc(Signal,('B','ZamkyDbf!','zámek věty 3 při cizím zámku'));
b:=false;
with locked TZAMD[4] do b:=true else b:=false;
if ^b then proc(Signal,('B','ZamkyDbf!','zámek věty 4 při cizím zámku věty 3'));
proc(Signal,('B','ZamkyDbf-veta-ok',''));
proc(Cekej,('A','ZamkyDbf-veta-konec',ok));
b:=false;
with locked TZAMD[3] do b:=true else b:=false;
if ^b then proc(Signal,('B','ZamkyDbf!','zámek věty 3 po uvolnění'));
proc(Signal,('B','ZamkyDbf-konec',''));
end;
