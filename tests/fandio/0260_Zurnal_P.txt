{ žurnál editace (journalof, edit(...,journal=)): oprava věty (O, N), zrušení (-),
  nová věta (+); číslo věty, uživatel, čas, údaje; údaj typu T jako číslo (pozice textu) }
VAR r:record of TZDAT; j:record of TZJ; i,p1,p2,p3,p4:real; chyba,t:string;
begin
chyba:='';
TZDAT.nrecs:=0;
TZJ.nrecs:=0;
for i:=1 to 3 do begin
  readrec(r,0);
  r.Kod:=i;
  r.Nazev:='Věta '+str(i,1,0);
  r.Pozn:='text '+str(i,1,0);
  writerec(r,0);
end;
{ oprava názvu věty 1, dolů, Ctrl-Y (věta 2), Ctrl-N a nová věta, Esc }
setkeybuf(char(13)+'Novy'+char(13)+char(0)+char(80)+char(25)+char(14)+'7'+char(13)+'Vlozena'+char(13)+char(13)+char(27));
edit(TZDAT,(),journal=TZJ);
t:='';
for i:=1 to TZJ.nrecs do begin
  readrec(j,i);
  t:=t+j.Upd+str(j.RecNr,1,0)+':'+str(j.Kod,1,0)+':'+trailchar(' ',j.Nazev)+' ';
  if (chyba='') & ((j.TimeStamp<today) | (j.TimeStamp>today+1)) then chyba:='čas věty žurnálu '+str(i,1,0);
end;
if (chyba='') & (t<>'O1:1:Věta 1 N1:1:Novy -2:2:Věta 2 +2:7:Vlozena ') then chyba:='žurnál: '+t;
{ údaj Pozn je v žurnálu číslo: pozice textu v souboru TZDAT }
if (chyba='') & (TZJ.nrecs=4) then begin
  readrec(j,1); p1:=j.Pozn;
  readrec(j,2); p2:=j.Pozn;
  readrec(j,3); p3:=j.Pozn;
  readrec(j,4); p4:=j.Pozn;
  if (p1<=0) | (p1<>p2) | (p3<=0) | (p3=p1) | (p4<>0) then
    chyba:='pozice textů v žurnálu '+str(p1,1,0)+' '+str(p2,1,0)+' '+str(p3,1,0)+' '+str(p4,1,0);
end;
proc(Vysl,('Zurnal',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
