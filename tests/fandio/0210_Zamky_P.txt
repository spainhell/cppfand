{ zámky souboru TZAM proti druhému uživateli (Zamky2): režim souboru (with shared)
  a zámek věty (with locked) }
VAR r:record of TZAM; i:real; ok:boolean; chyba:string;
begin
chyba:='';
TZAM.nrecs:=0;
for i:=1 to 5 do begin
  readrec(r,0);
  r.Kod:=i;
  writerec(r,0);
end;
{ výhradní režim: druhý uživatel nesmí číst ani zapisovat }
with shared TZAM(EXCL) do begin
  proc(Signal,('A','Zamky-excl',''));
  proc(Cekej,('B','Zamky-excl-ok',ok));
  if ^ok then chyba:='B neodpověděl při EXCL';
end
else chyba:='EXCL nezískán';
proc(Signal,('A','Zamky-excl-konec',''));
proc(Cekej,('B','Zamky-excl-konec-ok',ok));
{ čtení: druhý uživatel smí také číst, nesmí zapisovat }
with shared TZAM(RD) do begin
  proc(Signal,('A','Zamky-rd',''));
  proc(Cekej,('B','Zamky-rd-ok',ok));
end
else if chyba='' then chyba:='RD nezískán';
proc(Signal,('A','Zamky-rd-konec',''));
{ zámek věty 3 }
with locked TZAM[3] do begin
  proc(Signal,('A','Zamky-veta',''));
  proc(Cekej,('B','Zamky-veta-ok',ok));
end
else if chyba='' then chyba:='zámek věty 3 nezískán';
proc(Signal,('A','Zamky-veta-konec',''));
proc(Cekej,('B','Zamky-konec',ok));
if (chyba='') & ^ok then chyba:='druhý uživatel nedoběhl';
proc(ChybaB,('Zamky',chyba));
proc(Vysl,('Zamky',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
