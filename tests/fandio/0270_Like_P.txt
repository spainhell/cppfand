{ like: soubor se stejnou deklarací a rozšířením, like indexovaného souboru s prefixem
  klíčů; každý má vlastní data, texty a index, nemíchají se s originálem }
VAR a:record of TTXT; b:record of TLIKE; x:record of TIDX; y:record of TLIKX;
    i,p:real; chyba,t:string;
begin
chyba:='';
TTXT.nrecs:=0;
TLIKE.nrecs:=0;
for i:=1 to 3 do begin
  readrec(a,0);
  a.Id:=i;
  a.Pozn:='originál '+str(i,1,0);
  writerec(a,0);
end;
for i:=1 to 5 do begin
  readrec(b,0);
  b.Id:=10+i;
  b.Pozn:=repeatstr('L',100*i);
  b.Navic:='N'+str(i,1,0);
  writerec(b,0);
end;
close(TTXT);
close(TLIKE);
if (TTXT.nrecs<>3) | (TLIKE.nrecs<>5) then chyba:='nrecs '+str(TTXT.nrecs,1,0)+'/'+str(TLIKE.nrecs,1,0);
for i:=1 to TTXT.nrecs do begin
  readrec(a,i);
  if (chyba='') & ((a.Id<>i) | (a.Pozn<>'originál '+str(i,1,0))) then chyba:='originál věta '+str(i,1,0);
end;
for i:=1 to TLIKE.nrecs do begin
  readrec(b,i);
  if (chyba='') & ((b.Id<>10+i) | (b.Pozn<>repeatstr('L',100*i)) | (trailchar(' ',b.Navic)<>'N'+str(i,1,0))) then chyba:='like věta '+str(i,1,0);
end;
{ like indexovaného souboru: vlastní index, klíč LX_PodlePor }
TIDX.nrecs:=0;
TLIKX.nrecs:=0;
for i:=1 to 4 do begin
  readrec(x,0);
  x.Kod:='X'+str(i,1,0);
  x.Poradi:=i;
  writerec(x,0);
end;
for i:=1 to 6 do begin
  readrec(y,0);
  y.Kod:='Y'+str(7-i,1,0);
  y.Poradi:=i*10;
  writerec(y,0);
end;
t:='';
forall y/@ do t:=t+trailchar(' ',y.Kod)+' ';
if (chyba='') & (t<>'Y1 Y2 Y3 Y4 Y5 Y6 ') then chyba:='TLIKX podle  '+t;
t:='';
forall y/LX_PodlePor do t:=t+str(y.Poradi,1,0)+' ';
if (chyba='') & (t<>'60 50 40 30 20 10 ') then chyba:='TLIKX podle LX_PodlePor: '+t;
t:='';
forall x/@ do t:=t+trailchar(' ',x.Kod)+' ';
if (chyba='') & (t<>'X1 X2 X3 X4 ') then chyba:='TIDX po zápisu do TLIKX: '+t;
proc(Vysl,('Like',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
