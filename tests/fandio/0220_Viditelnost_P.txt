{ změny druhého uživatele (Viditelnost2) jsou vidět: počet vět, přepsaná, přidaná
  a zrušená věta, texty; přidaný text nesmí přepsat texty druhého uživatele }
VAR r:record of TVID; i,k:real; ok:boolean; chyba,t:string;
begin
chyba:='';
TVID.nrecs:=0;
for i:=1 to 10 do begin
  readrec(r,0);
  r.Kod:=i;
  r.Nazev:='A'+str(i,1,0);
  r.Pozn:=repeatstr(char(64+i),100*i);
  writerec(r,0);
end;
proc(Signal,('A','vid-zapsano',''));
proc(Cekej,('B','vid-zmeneno',ok));
if ^ok then chyba:='druhý uživatel nezměnil soubor';
{ B přepsal větu 5, přidal větu s Kod 11 a zrušil větu 2 }
if (chyba='') & (TVID.nrecs<>10) then chyba:='nrecs po změnách B: '+str(TVID.nrecs,1,0);
for i:=1 to TVID.nrecs do begin
  readrec(r,i);
  k:=cond((i=1):1,(i=10):11,else:i+1);
  t:=cond((k=5):repeatstr('b',3000),(k=11):repeatstr('c',2000),else:repeatstr(char(64+k),100*k));
  if (chyba='') & ((r.Kod<>k) | (r.Pozn<>t)) then chyba:='po změnách B věta '+str(i,1,0)+' Kod '+str(r.Kod,1,0);
end;
readrec(r,4);
if (chyba='') & (trailchar(' ',r.Nazev)<>'B5') then chyba:='přepsaný název věty Kod 5: '+r.Nazev;
{ nový text po textech B }
readrec(r,0);
r.Kod:=12;
r.Pozn:=repeatstr('d',1500);
writerec(r,0);
proc(Signal,('A','vid-pridano',''));
proc(Cekej,('B','Viditelnost-konec',ok));
if (chyba='') & ^ok then chyba:='druhý uživatel nedoběhl';
proc(ChybaB,('Viditelnost',chyba));
proc(Vysl,('Viditelnost',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
