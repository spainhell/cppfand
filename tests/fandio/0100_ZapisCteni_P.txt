{ zápis a čtení vět neindexovaného souboru, přepis vět, čtení po zavření souboru }
VAR r:record of TDAT; i,d0:real; chyba:string;
begin
chyba:='';
TDAT.nrecs:=0;
d0:=valdate('01.01.2024','DD.MM.YYYY');
for i:=1 to 200 do begin
  readrec(r,0);
  r.Cislo:=i;
  r.Text:='Věta '+str(i,3,0);
  r.Kod:=str(100000+i,6,0);
  r.Castka:=i*1.25;
  r.Realne:=i/3;
  r.Datum:=d0+i;
  r.Priznak:=(i-int(i/2)*2)=0;
  writerec(r,0);
end;
if TDAT.nrecs<>200 then chyba:='nrecs='+str(TDAT.nrecs,1,0);
{ přepis každé desáté věty }
for i:=1 to 20 do begin
  readrec(r,i*10);
  r.Castka:=-r.Castka;
  writerec(r,i*10);
end;
close(TDAT);
for i:=1 to 200 do begin
  readrec(r,i);
  if chyba='' then begin
    if r.Cislo<>i then chyba:='Cislo, věta '+str(i,1,0)
    else if trailchar(' ',r.Text)<>'Věta '+str(i,3,0) then chyba:='Text, věta '+str(i,1,0)
    else if r.Kod<>str(100000+i,6,0) then chyba:='Kod, věta '+str(i,1,0)
    else if r.Castka<>cond(((i-int(i/10)*10)=0):-i*1.25,else:i*1.25) then chyba:='Castka, věta '+str(i,1,0)
    else if abs(r.Realne-i/3)>0.0000001 then chyba:='Realne, věta '+str(i,1,0)
    else if r.Datum<>d0+i then chyba:='Datum, věta '+str(i,1,0)
    else if (r.Priznak & ((i-int(i/2)*2)<>0)) | (^r.Priznak & ((i-int(i/2)*2)=0)) then chyba:='Priznak, věta '+str(i,1,0);
  end;
end;
proc(Vysl,('ZapisCteni',cond(chyba='':'OK',else:'CHYBA'),chyba));
end;
